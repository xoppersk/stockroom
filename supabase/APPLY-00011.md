## The migration (paste this whole block)

```sql
-- =============================================================================
-- 00011_production.sql — production readiness: refunds, store settings,
-- customer order ownership
--
--   * public.issue_order_refund() — the SECURITY DEFINER refund path promised
--     by 00008's money-guard comment: admin/support refund without touching
--     money columns directly (the guard trigger is bypassed transaction-locally,
--     the same pattern handle_stripe_event uses).
--   * public.store_settings — key/value (jsonb) store settings: tax_rate,
--     shipping_flat, free_shipping_threshold. Staff read, admin write.
--   * create_order_with_items() + create_checkout_session() — re-created with
--     settings-driven tax/shipping totals. Defaults (0 / 0 / null) reproduce
--     the pre-00011 totals exactly, so behaviour only changes once an admin
--     edits the settings. create_checkout_session() also now records
--     created_by = auth.uid() on the order (00008's staff path already did).
--   * Customer owner-read policies: a signed-in customer reads their own
--     orders / order_items / order_events (orders.created_by = auth.uid()).
--     Additive only — no existing policy is weakened or removed.
--   * public.set_low_stock_threshold() — SECURITY DEFINER threshold setter
--     (admin/warehouse), the write path inventory_levels deliberately lacks.
--
-- Idempotent: create-or-replace functions, drop-if-exists policies, seed rows
-- use ON CONFLICT DO NOTHING (an admin's edited settings survive a re-run).
-- =============================================================================

-- ---------------------------------------------------------------------------
-- store_settings
-- ---------------------------------------------------------------------------
create table if not exists public.store_settings (
  key        text primary key,
  value      jsonb not null,
  updated_at timestamptz not null default now()
);

comment on table public.store_settings is
  'Store-wide settings as jsonb values. Read by create_order_with_items() / create_checkout_session(); written by admins only.';

drop trigger if exists store_settings_set_updated_at on public.store_settings;
create trigger store_settings_set_updated_at
  before update on public.store_settings
  for each row execute function public.set_updated_at();

-- Seed defaults: 0 / 0 / null reproduce today's totals exactly.
insert into public.store_settings (key, value) values
  ('tax_rate', '0'::jsonb),
  ('shipping_flat', '0'::jsonb),
  ('free_shipping_threshold', 'null'::jsonb)
on conflict (key) do nothing;

alter table public.store_settings enable row level security;

-- Staff read; admin write only.
drop policy if exists store_settings_staff_read on public.store_settings;
create policy store_settings_staff_read
  on public.store_settings for select to authenticated
  using (public.is_staff());

drop policy if exists store_settings_admin_write on public.store_settings;
create policy store_settings_admin_write
  on public.store_settings for all to authenticated
  using (public.is_admin())
  with check (public.is_admin());

grant select on public.store_settings to authenticated;
grant insert, update, delete on public.store_settings to authenticated;

-- ---------------------------------------------------------------------------
-- _store_setting_numeric(p_key): internal reader for the totals computation.
-- Returns the setting as numeric, or NULL when the row is missing or the
-- jsonb value is not a number (e.g. free_shipping_threshold = null).
-- ---------------------------------------------------------------------------
create or replace function public._store_setting_numeric(p_key text)
returns numeric
language sql
stable
security definer
set search_path = public
as $$
  select case
    when jsonb_typeof(value) = 'number' then (value #>> '{}')::numeric
    else null
  end
  from public.store_settings
  where key = p_key;
$$;

comment on function public._store_setting_numeric(text) is
  'Internal: numeric store_settings value for a key, NULL when unset/non-numeric.';

revoke all on function public._store_setting_numeric(text) from public, anon;
grant execute on function public._store_setting_numeric(text) to authenticated;

-- ---------------------------------------------------------------------------
-- issue_order_refund(p_order_id, p_amount, p_reason)
-- The refund path 00008 promised: the money-guard trigger blocks non-admin
-- changes to refunded_total, so refunds run through this SECURITY DEFINER
-- function (admin/support), which sets the same transaction-local bypass
-- handle_stripe_event() uses, validates the amount against the remaining
-- refundable balance, moves payment_status (and status, only on a full
-- refund), and writes order_events + notification + audit_log.
-- Returns the order's new refunded_total.
-- ---------------------------------------------------------------------------
create or replace function public.issue_order_refund(
  p_order_id uuid,
  p_amount numeric,
  p_reason text default null
)
returns numeric
language plpgsql
security definer
set search_path = public
as $$
declare
  v_role         text := public.current_role();
  v_order        public.orders%rowtype;
  v_amount       numeric(12,2) := round(p_amount, 2);
  v_new_refunded numeric(12,2);
  v_fully        boolean;
begin
  if v_role is distinct from 'admin' and v_role is distinct from 'support' then
    raise exception 'Only admins and support staff can issue refunds.';
  end if;

  if p_amount is null or v_amount <= 0 then
    raise exception 'Refund amount must be greater than zero.';
  end if;

  select * into v_order
    from public.orders
   where id = p_order_id
   for update;

  if not found then
    raise exception 'Order % does not exist.', p_order_id;
  end if;

  v_new_refunded := v_order.refunded_total + v_amount;

  if v_new_refunded > v_order.total then
    raise exception 'Refund of % exceeds the remaining refundable amount (%).',
      v_amount, v_order.total - v_order.refunded_total;
  end if;

  v_fully := v_new_refunded = v_order.total;

  -- Money columns move here; the guard is bypassed transaction-locally,
  -- exactly as handle_stripe_event() does for webhook-driven refunds.
  perform set_config('app.money_guard_bypass', 'on', true);

  update public.orders
     set refunded_total = v_new_refunded,
         payment_status = case when v_fully then 'refunded' else 'partially_refunded' end,
         -- status becomes 'refunded' only on a full refund; a partial refund
         -- leaves the fulfillment status (paid/fulfilled/…) untouched.
         status = case when v_fully then 'refunded' else status end,
         updated_at = now()
   where id = p_order_id;

  insert into public.order_events (order_id, event_type, message, metadata, created_by)
  values (
    p_order_id, 'refund_issued',
    'Refund issued for ' || v_order.order_number || ' ($' || v_amount::text || ').',
    jsonb_build_object(
      'amount', v_amount,
      'refunded_total', v_new_refunded,
      'reason', p_reason
    ),
    auth.uid()
  );

  insert into public.notifications (user_id, kind, title, body, link)
  values (
    null, 'refund_issued',
    'Refund issued: ' || v_order.order_number,
    '$' || v_amount::text || ' refunded.',
    '/orders/' || v_order.order_number
  );

  insert into public.audit_log (actor_id, action, table_name, row_id, after)
  values (
    auth.uid(), 'order.refund', 'orders', p_order_id::text,
    jsonb_build_object(
      'order_number', v_order.order_number,
      'amount', v_amount,
      'refunded_total', v_new_refunded,
      'reason', p_reason
    )
  );

  return v_new_refunded;
end;
$$;

comment on function public.issue_order_refund(uuid, numeric, text) is
  'Admin/support refund (SECURITY DEFINER): validates against the remaining refundable balance, updates refunded_total + payment_status (status only on full refund), writes timeline + notification + audit. Returns the new refunded_total.';

revoke all on function public.issue_order_refund(uuid, numeric, text) from public, anon;
grant execute on function public.issue_order_refund(uuid, numeric, text) to authenticated;

-- ---------------------------------------------------------------------------
-- set_low_stock_threshold(p_variant_id, p_threshold)
-- inventory_levels has no direct write policies (00006); this is the
-- role-checked setter for the low-stock threshold. Admin/warehouse, mirroring
-- adjust_inventory(). Raises when the variant has no inventory row.
-- ---------------------------------------------------------------------------
create or replace function public.set_low_stock_threshold(
  p_variant_id uuid,
  p_threshold integer
)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_before integer;
begin
  if not (public.is_admin() or public.current_role() = 'warehouse') then
    raise exception 'Only warehouse and admin staff can set low-stock thresholds.';
  end if;

  if p_threshold is null or p_threshold < 0 then
    raise exception 'Low-stock threshold must be zero or greater.';
  end if;

  select low_stock_threshold into v_before
    from public.inventory_levels
   where variant_id = p_variant_id
   for update;

  if not found then
    raise exception 'No inventory record for variant %.', p_variant_id;
  end if;

  update public.inventory_levels
     set low_stock_threshold = p_threshold,
         updated_at = now()
   where variant_id = p_variant_id;

  insert into public.audit_log (actor_id, action, table_name, row_id, before, after)
  values (
    auth.uid(),
    'inventory.set_low_stock_threshold',
    'inventory_levels',
    p_variant_id::text,
    jsonb_build_object('low_stock_threshold', v_before),
    jsonb_build_object('low_stock_threshold', p_threshold)
  );
end;
$$;

comment on function public.set_low_stock_threshold(uuid, integer) is
  'Role-checked (admin/warehouse) setter for inventory_levels.low_stock_threshold. Raises when the variant has no inventory row.';

revoke all on function public.set_low_stock_threshold(uuid, integer) from public, anon;
grant execute on function public.set_low_stock_threshold(uuid, integer) to authenticated;

-- ---------------------------------------------------------------------------
-- create_order_with_items(): re-created from 00008 with settings-driven
-- tax/shipping totals. Everything else is byte-for-byte the 00008 behaviour.
--   tax_total      = round((subtotal - discount_total) * tax_rate, 2)
--   shipping_total = 0 when free_shipping_threshold is set and the discounted
--                    subtotal reaches it, else shipping_flat
--   total          = subtotal - discount_total + tax_total + shipping_total
-- ---------------------------------------------------------------------------
create or replace function public.create_order_with_items(
  p_customer_id uuid,
  p_items jsonb,
  p_discount_code text default null,
  p_shipping_address jsonb default null,
  p_source text default 'manual',
  p_note text default null
)
returns uuid
language plpgsql
security definer
set search_path = public
as $$
declare
  v_role           text := public.current_role();
  v_item           jsonb;
  v_variant_id     uuid;
  v_qty            integer;
  v_price          numeric(12,2);
  v_variant_title  text;
  v_product_title  text;
  v_product_status text;
  v_sku            text;
  v_stock          integer;
  v_subtotal       numeric(12,2) := 0;
  v_discount       public.discounts%rowtype;
  v_discount_total numeric(12,2) := 0;
  v_taxable        numeric(12,2);
  v_tax_total      numeric(12,2) := 0;
  v_shipping_total numeric(12,2) := 0;
  v_total          numeric(12,2);
  v_order_id       uuid;
  v_order_number   text;
begin
  if v_role is distinct from 'admin' and v_role is distinct from 'support' then
    raise exception 'Only admins and support staff can create orders.';
  end if;

  if p_items is null
     or jsonb_typeof(p_items) <> 'array'
     or jsonb_array_length(p_items) = 0 then
    raise exception 'An order needs at least one line item.';
  end if;

  -- Validate + price every line, locking inventory rows up front so two
  -- concurrent orders cannot oversell the same variant.
  for v_item in select * from jsonb_array_elements(p_items)
  loop
    begin
      v_variant_id := (v_item ->> 'variant_id')::uuid;
      v_qty := (v_item ->> 'quantity')::integer;
    exception when others then
      raise exception 'Each line item needs a variant_id (uuid) and a quantity (integer).';
    end;

    if v_qty is null or v_qty <= 0 then
      raise exception 'Quantity must be a positive integer.';
    end if;

    select pv.price, pv.title, pv.sku, p.title, p.status
      into v_price, v_variant_title, v_sku, v_product_title, v_product_status
      from public.product_variants pv
      join public.products p on p.id = pv.product_id
     where pv.id = v_variant_id;

    if not found then
      raise exception 'Variant % does not exist.', v_variant_id;
    end if;
    if v_product_status is distinct from 'published' then
      raise exception 'Product "%" is not published.', v_product_title;
    end if;

    select quantity_on_hand into v_stock
      from public.inventory_levels
     where variant_id = v_variant_id
     for update;
    if not found then
      raise exception 'No inventory record for variant %.', v_variant_id;
    end if;
    if v_stock < v_qty then
      raise exception 'Insufficient stock for % (%): have %, need %.',
        v_variant_title, v_sku, v_stock, v_qty;
    end if;

    v_subtotal := v_subtotal + v_price * v_qty;
  end loop;

  -- Discount: validated server-side; the client never dictates the amount.
  if p_discount_code is not null and btrim(p_discount_code) <> '' then
    v_discount := public.apply_discount_validation(p_discount_code, v_subtotal, p_customer_id);
    if v_discount.kind = 'percentage' then
      v_discount_total := round(v_subtotal * v_discount.value / 100, 2);
    else
      v_discount_total := least(v_discount.value, v_subtotal);
    end if;
  end if;

  -- Tax + shipping from store_settings (00011). Defaults are 0 / 0 / null,
  -- which reproduce the pre-00011 totals exactly.
  v_taxable := v_subtotal - v_discount_total;
  v_tax_total := round(v_taxable * coalesce(public._store_setting_numeric('tax_rate'), 0), 2);
  v_shipping_total := case
    when public._store_setting_numeric('free_shipping_threshold') is not null
         and v_taxable >= public._store_setting_numeric('free_shipping_threshold')
      then 0::numeric(12,2)
    else coalesce(public._store_setting_numeric('shipping_flat'), 0)::numeric(12,2)
  end;

  v_total := v_taxable + v_tax_total + v_shipping_total;

  v_order_number := 'ORD-' || nextval('public.order_number_seq');

  insert into public.orders (
    order_number, customer_id, status, payment_status,
    subtotal, discount_total, tax_total, shipping_total, total,
    currency, discount_code, shipping_address, source, created_by
  ) values (
    v_order_number, p_customer_id, 'pending', 'pending',
    v_subtotal, v_discount_total, v_tax_total, v_shipping_total, v_total,
    'USD', nullif(upper(btrim(coalesce(p_discount_code, ''))), ''),
    p_shipping_address, p_source, auth.uid()
  )
  returning id into v_order_id;

  -- Line items (snapshots) + stock decrement with 'sale' adjustments.
  for v_item in select * from jsonb_array_elements(p_items)
  loop
    v_variant_id := (v_item ->> 'variant_id')::uuid;
    v_qty := (v_item ->> 'quantity')::integer;

    select pv.price, pv.title, pv.sku, p.title
      into v_price, v_variant_title, v_sku, v_product_title
      from public.product_variants pv
      join public.products p on p.id = pv.product_id
     where pv.id = v_variant_id;

    insert into public.order_items (
      order_id, variant_id, product_title, variant_title, sku,
      quantity, unit_price, line_total
    ) values (
      v_order_id, v_variant_id, v_product_title, v_variant_title, v_sku,
      v_qty, v_price, v_price * v_qty
    );

    perform public._adjust_inventory(
      v_variant_id, -v_qty, 'sale', v_order_number,
      'Order ' || v_order_number, auth.uid()
    );
  end loop;

  if v_discount.id is not null then
    insert into public.discount_redemptions (discount_id, order_id, customer_id, amount)
    values (v_discount.id, v_order_id, p_customer_id, v_discount_total);
  end if;

  insert into public.order_events (order_id, event_type, message, created_by)
  values (
    v_order_id, 'created',
    'Order ' || v_order_number || ' created (' || p_source || ').',
    auth.uid()
  );

  if p_note is not null and btrim(p_note) <> '' then
    insert into public.order_events (order_id, event_type, message, created_by)
    values (v_order_id, 'note_added', p_note, auth.uid());
  end if;

  insert into public.notifications (user_id, kind, title, body, link)
  values (
    null, 'new_order',
    'New order ' || v_order_number,
    'Total $' || v_total::text || ' (' || v_subtotal::text || ' before discounts).',
    '/orders/' || v_order_number
  );

  insert into public.audit_log (actor_id, action, table_name, row_id, after)
  values (
    auth.uid(), 'order.create', 'orders', v_order_id::text,
    jsonb_build_object('order_number', v_order_number, 'total', v_total)
  );

  return v_order_id;
end;
$$;

comment on function public.create_order_with_items(uuid, jsonb, text, jsonb, text, text) is
  'Staff order creation (admin/support only): validates discount server-side, prices from live variants, applies store_settings tax/shipping, decrements stock, writes timeline + notification + audit.';

-- ---------------------------------------------------------------------------
-- create_checkout_session(): re-created from 00009 with settings-driven
-- tax/shipping totals (identical formula to create_order_with_items) and
-- created_by = auth.uid() on the order insert. Everything else is
-- byte-for-byte the 00009 behaviour.
-- ---------------------------------------------------------------------------
create or replace function public.create_checkout_session(
  p_cart_id uuid,
  p_guest_token uuid
)
returns uuid
language plpgsql
security definer
set search_path = public
as $$
declare
  v_cart           public.carts%rowtype;
  v_variant_id     uuid;
  v_qty            integer;
  v_price          numeric(12,2);
  v_variant_title  text;
  v_product_title  text;
  v_product_status text;
  v_sku            text;
  v_subtotal       numeric(12,2) := 0;
  v_discount       public.discounts%rowtype;
  v_discount_total numeric(12,2) := 0;
  v_taxable        numeric(12,2);
  v_tax_total      numeric(12,2) := 0;
  v_shipping_total numeric(12,2) := 0;
  v_total          numeric(12,2);
  v_order_id       uuid;
  v_order_number   text;
begin
  select * into v_cart from public.carts where id = p_cart_id;
  if not found then
    raise exception 'Cart not found.';
  end if;

  -- Cart-token check: the token must match, or the cart must belong to the
  -- signed-in shopper. Same "not found" message either way (no oracle).
  if not (
    v_cart.guest_token = p_guest_token
    or (v_cart.user_id is not null and v_cart.user_id = (select auth.uid()))
  ) then
    raise exception 'Cart not found.';
  end if;

  if v_cart.expires_at < now() then
    raise exception 'This cart has expired.';
  end if;

  -- Lock the cart's lines for the duration of the checkout.
  perform 1 from public.cart_items where cart_id = p_cart_id for update;

  for v_variant_id, v_qty in
    select ci.variant_id, ci.quantity
      from public.cart_items ci
     where ci.cart_id = p_cart_id
     order by ci.added_at
  loop
    select pv.price, pv.title, pv.sku, p.title, p.status
      into v_price, v_variant_title, v_sku, v_product_title, v_product_status
      from public.product_variants pv
      join public.products p on p.id = pv.product_id
     where pv.id = v_variant_id;

    if not found then
      raise exception 'An item in your cart is no longer available.';
    end if;
    if v_product_status is distinct from 'published' then
      raise exception 'Product "%" is no longer available.', v_product_title;
    end if;

    -- Availability check only — the decrement happens on payment success.
    perform 1 from public.inventory_levels
     where variant_id = v_variant_id and quantity_on_hand >= v_qty;
    if not found then
      raise exception 'Insufficient stock for % (%).', v_variant_title, v_sku;
    end if;

    v_subtotal := v_subtotal + v_price * v_qty;
  end loop;

  if v_subtotal = 0 then
    raise exception 'Your cart is empty.';
  end if;

  -- Discount re-validated server-side at checkout time.
  if v_cart.discount_code is not null and btrim(v_cart.discount_code) <> '' then
    v_discount := public.apply_discount_validation(v_cart.discount_code, v_subtotal, null);
    if v_discount.kind = 'percentage' then
      v_discount_total := round(v_subtotal * v_discount.value / 100, 2);
    else
      v_discount_total := least(v_discount.value, v_subtotal);
    end if;
  end if;

  -- Tax + shipping from store_settings (00011) — same formula as
  -- create_order_with_items. Defaults reproduce the pre-00011 totals.
  v_taxable := v_subtotal - v_discount_total;
  v_tax_total := round(v_taxable * coalesce(public._store_setting_numeric('tax_rate'), 0), 2);
  v_shipping_total := case
    when public._store_setting_numeric('free_shipping_threshold') is not null
         and v_taxable >= public._store_setting_numeric('free_shipping_threshold')
      then 0::numeric(12,2)
    else coalesce(public._store_setting_numeric('shipping_flat'), 0)::numeric(12,2)
  end;

  v_total := v_taxable + v_tax_total + v_shipping_total;
  v_order_number := 'ORD-' || nextval('public.order_number_seq');

  insert into public.orders (
    order_number, customer_id, status, payment_status,
    subtotal, discount_total, tax_total, shipping_total, total,
    currency, discount_code, guest_token, source, created_by
  ) values (
    v_order_number, null, 'pending', 'pending',
    v_subtotal, v_discount_total, v_tax_total, v_shipping_total, v_total,
    'USD', nullif(upper(btrim(coalesce(v_cart.discount_code, ''))), ''),
    p_guest_token, 'storefront', auth.uid()
  )
  returning id into v_order_id;

  for v_variant_id, v_qty in
    select ci.variant_id, ci.quantity
      from public.cart_items ci
     where ci.cart_id = p_cart_id
     order by ci.added_at
  loop
    select pv.price, pv.title, pv.sku, p.title
      into v_price, v_variant_title, v_sku, v_product_title
      from public.product_variants pv
      join public.products p on p.id = pv.product_id
     where pv.id = v_variant_id;

    insert into public.order_items (
      order_id, variant_id, product_title, variant_title, sku,
      quantity, unit_price, line_total
    ) values (
      v_order_id, v_variant_id, v_product_title, v_variant_title, v_sku,
      v_qty, v_price, v_price * v_qty
    );
  end loop;

  insert into public.payments (
    order_id, stripe_payment_intent_id, amount, currency, status, idempotency_key
  ) values (
    v_order_id, null, v_total, 'USD', 'pending', 'order:' || v_order_id::text
  );

  if v_discount.id is not null then
    insert into public.discount_redemptions (discount_id, order_id, customer_id, amount)
    values (v_discount.id, v_order_id, null, v_discount_total);
  end if;

  insert into public.order_events (order_id, event_type, message)
  values (
    v_order_id, 'checkout_started',
    'Checkout started for ' || v_order_number || ' (storefront).'
  );

  return v_order_id;
end;
```
